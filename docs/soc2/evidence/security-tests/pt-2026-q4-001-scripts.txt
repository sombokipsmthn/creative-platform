## Test Scripts Used

### 1. Authentication Bypass Test Script
```bash
#!/bin/bash
# Test unauthenticated access to protected endpoints

ENDPOINTS=(
  "/api/clients"
  "/api/quotes"
  "/api/invoices"
  "/api/galleries/upload"
)

for endpoint in "${ENDPOINTS[@]}"; do
  echo "Testing $endpoint without authentication"
  curl -i -X GET "http://localhost:3005$endpoint"
  echo ""
done
```

### 3. Input Validation Test Script
```bash
#!/bin/bash
# Test input validation with malicious payloads

PAYLOADS=(
  "' OR '1'='1"
  "<script>alert('xss')</script>"
  "'; DROP TABLE users;--"
  "../../etc/passwd"
  "; ls -la"
  "| cat /etc/passwd"
)

ENDPOINT="/api/clients"

for payload in "${PAYLOADS[@]}"; do
  echo "Testing payload: $payload"
  curl -i -X POST "http://localhost:3005$endpoint" \
    -H "Content-Type: application/json" \
    -d "{\"name\":\"$payload\"}"
  echo ""
done
```

### 4. Rate Limiting Test Script
